% ================= SECTION 1: Introduction ================= %
\section{Introduction to JAX}

\begin{frame}{What is JAX?}
    \textbf{JAX} is a high-performance numerical computing library developed by Google Research. It brings together three powerful concepts:

    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{NumPy API:} JAX provides a familiar \texttt{jax.numpy} interface, acting as a drop-in replacement for standard NumPy.
        \item \textbf{Autograd:} It allows for automatic differentiation of native Python and NumPy code.
        \item \textbf{XLA (Accelerated Linear Algebra):} A domain-specific compiler that accelerates math operations across CPUs, GPUs, and TPUs.
    \end{itemize}

    \vspace{0.3cm}
    \textit{Goal: Allow researchers to write simple Python code that scales effortlessly to massive hardware accelerators.}
\end{frame}

\begin{frame}{JAX vs. Standard NumPy}
    While \texttt{jax.numpy} mirrors the \texttt{numpy} API, it has several superpowers under the hood.

    \vspace{0.5cm}
    \begin{table}[]
        \centering
        \begin{tabular}{@{} l cc @{}}
            \toprule
            \textbf{Feature} & \textbf{Standard NumPy} & \textbf{JAX}              \\
            \midrule
            Hardware Support & CPU only                & CPU, GPU, TPU             \\
            Execution Model  & Eager                   & Eager \& Compiled (JIT)   \\
            Derivatives      & Manual / Finite Diff    & Automatic (\texttt{grad}) \\
            Vectorization    & Manual batching         & Automatic (\texttt{vmap}) \\
            Mutability       & Mutable arrays          & \textbf{Immutable arrays} \\
            \bottomrule
        \end{tabular}
        \caption{Comparison of features between NumPy and JAX.}
    \end{table}
\end{frame}

% ================= SECTION 2: JIT ================= %
\section{Just-In-Time Compilation (JIT)}

\begin{frame}{Just-In-Time Compilation (\texttt{jax.jit})}
    \textbf{Problem:} Standard Python executes line-by-line (eagerly). This introduces significant overhead, especially for complex mathematical functions.

    \vspace{0.3cm}
    \textbf{Solution:} \texttt{jax.jit} traces your Python function and compiles it into highly optimized machine code using the \textbf{XLA} compiler.

    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Operation Fusion:} Multiple mathematical operations are merged into a single kernel, avoiding memory read/write bottlenecks.
        \item \textbf{Hardware Optimization:} XLA optimizes the code specifically for the hardware you are running on (GPU/TPU).
    \end{itemize}
\end{frame}

\begin{frame}{Schematic: How \texttt{jit} Works}
    \begin{center}
        \begin{tikzpicture}[node distance=2.5cm]
            \node (python) [process] {Python Function \\ \texttt{def f(x):}};
            \node (tracer) [io, right of=python, xshift=2cm] {JAX Tracer};
            \node (jaxpr) [process, right of=tracer, xshift=3cm] {JAXpr \\ (Intermediate Rep)};
            \node (xla) [decision, below of=jaxpr, yshift=-0.5cm] {XLA Compiler};
            \node (machine) [process, left of=xla, xshift=-4cm] {Optimized Machine Code \\ (GPU/TPU Executable)};

            \draw [arrow] (python) -- (tracer);
            \draw [arrow] (tracer) -- node[anchor=south] {Trace} (jaxpr);
            \draw [arrow] (jaxpr) -- (xla);
            \draw [arrow] (xla) -- node[anchor=north] {Compile} (machine);
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \small \textit{Note: The compilation only happens once (on the first function call). Subsequent calls reuse the compiled executable instantly.}
\end{frame}

\begin{frame}[fragile]{Code Example: \texttt{jax.jit}}
    \begin{lstlisting}
import jax.numpy as jnp
from jax import jit
import time

def slow_selu(x, alpha=1.67, lmbda=1.05):
    # Standard Python execution
    return lmbda * jnp.where(x > 0, x, alpha * jnp.exp(x) - alpha)

# Compile the function
fast_selu = jit(slow_selu)

x = jnp.arange(1000000.0)
_ = fast_selu(x) # First run compiles (warmup)

# Second run is lightning fast!
res = fast_selu(x)
    \end{lstlisting}
\end{frame}

% ================= SECTION 3: VMAP ================= %
\section{Automatic Vectorization (VMAP)}

\begin{frame}{Automatic Vectorization (\texttt{jax.vmap})}
    \textbf{Problem:} Writing code that handles varying batch sizes usually involves complex, error-prone manual loops or reshaping operations.

    \vspace{0.3cm}
    \textbf{Solution:} \texttt{jax.vmap} allows you to write a function that operates on a \textit{single} data point. It then automatically transforms this function to operate on a \textit{batch} of data points.

    \vspace{0.3cm}
    \begin{itemize}
        \item Keeps code clean and mathematical.
        \item Pushes the loop down into the C++ / XLA level for performance.
    \end{itemize}
\end{frame}

\begin{frame}{Schematic: How \texttt{vmap} Works}
    \begin{center}
        \begin{tikzpicture}[node distance=1.5cm]
            % Single Execution
            \node (in1) [io, minimum width=1.5cm] {Vector $x$};
            \node (f1) [process, right of=in1, xshift=2cm] {\texttt{f(x)}};
            \node (out1) [io, right of=f1, xshift=2cm, minimum width=1.5cm] {Scalar $y$};

            \draw [arrow] (in1) -- (f1);
            \draw [arrow] (f1) -- (out1);

            % VMAP Execution
            \node (in2) [io, below of=in1, yshift=-1.5cm, minimum width=1.5cm, fill=secondaryColor!20] {Matrix $X$ \\ (Batch)};
            \node (vmapf) [process, right of=in2, xshift=2cm, fill=gray!20] {\texttt{vmap(f)}};
            \node (out2) [io, right of=vmapf, xshift=2cm, minimum width=1.5cm, fill=secondaryColor!20] {Vector $Y$ \\ (Batch)};

            \draw [arrow] (in2) -- (vmapf);
            \draw [arrow] (vmapf) -- (out2);

            % Label
            \node at (4.5, -1.2) {\textit{\textbf{Automatic Batching}}};
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \small \textit{\texttt{vmap} maps the logic across a new dimension without any manual \texttt{for} loops.}
\end{frame}

\begin{frame}[fragile]{Code Example: \texttt{jax.vmap}}
    \begin{lstlisting}
from jax import vmap
import jax.numpy as jnp

# Function strictly for 1D vectors
def dot_product(v1, v2):
    return jnp.dot(v1, v2)

# Create two 2D matrices (batches of 10 vectors of size 5)
mat1 = jnp.ones((10, 5))
mat2 = jnp.ones((10, 5))

# Automatically vectorize to handle the matrix inputs
# in_axes=(0, 0) maps over the first axis (rows)
v_dot = vmap(dot_product, in_axes=(0, 0))

result = v_dot(mat1, mat2) # Output shape: (10,)
    \end{lstlisting}
\end{frame}

% ================= SECTION 4: GRAD ================= %
\section{Automatic Differentiation (GRAD)}

\begin{frame}{Automatic Differentiation (\texttt{jax.grad})}
    \textbf{Problem:} Deriving gradients for complex loss functions in Machine Learning by hand is impossible, and finite differences are numerically unstable.

    \vspace{0.3cm}
    \textbf{Solution:} \texttt{jax.grad} provides exact, analytical gradients of scalar-valued functions using Reverse-Mode Automatic Differentiation.

    \vspace{0.3cm}
    \begin{itemize}
        \item Automatically computes $\frac{df}{dx}$.
        \item Can be chained for higher-order derivatives (\texttt{grad(grad(f))}).
    \end{itemize}
\end{frame}

\begin{frame}{Schematic: Forward and Backward Pass}
    \begin{center}
        \begin{tikzpicture}[node distance=2.5cm]
            \node (input) [io] {Input $x$};
            \node (fpass) [process, right of=input, xshift=1.5cm] {Compute $f(x)$ \\ (Forward Pass)};
            \node (output) [io, right of=fpass, xshift=1.5cm] {Output $y$};

            \node (gradf) [process, below of=fpass, fill=secondaryColor!10, draw=secondaryColor, text=secondaryColor, yshift=0.5cm] {\texttt{grad(f)}};
            \node (gradient) [io, left of=gradf, xshift=-1.5cm, fill=secondaryColor!10] {$\frac{df}{dx}$};

            \draw [arrow] (input) -- (fpass);
            \draw [arrow] (fpass) -- (output);

            \draw [thick,<-,>=stealth, secondaryColor, dashed] (gradf) -- node[anchor=west] {Reverse AD} (output);
            \draw [thick,->,>=stealth, secondaryColor] (gradf) -- (gradient);
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \small \textit{\texttt{grad} traces the computational graph built during the forward pass and applies the chain rule backward to compute exact derivatives.}
\end{frame}

\begin{frame}[fragile]{Code Example: \texttt{jax.grad}}
    \begin{lstlisting}
from jax import grad

# Define a polynomial: f(x) = 3x^2 + 2x + 1
def f(x):
    return 3 * x**2 + 2 * x + 1

# Automatically create the derivative function
# Analytical derivative: df/dx = 6x + 2
df_dx = grad(f)

x_val = 2.0
print("f(x)  =", f(x_val))       # Output: 17.0
print("f'(x) =", df_dx(x_val))   # Output: 14.0
    \end{lstlisting}
\end{frame}

% ================= SECTION 5: Pytrees ================= %
\section{Pytrees}

\begin{frame}{Working with Pytrees}
    \textbf{Problem:} Machine learning models have parameters stored in complex, nested dictionaries, lists, and tuples. Iterating through them manually is tedious.

    \vspace{0.3cm}
    \textbf{Solution:} JAX treats all nested Python structures as \textbf{Pytrees}.

    \vspace{0.3cm}
    \begin{itemize}
        \item A Pytree is a container of leaf elements.
        \item JAX provides \texttt{jax.tree\_map} to apply a function to every leaf in the entire nested structure instantly.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Code Example: Pytrees}
    \begin{lstlisting}
from jax.tree_util import tree_map

# A complex nested data structure representing model weights
params = {
    'layer1': [1.0, 2.0, 3.0],
    'layer2': {'w': 0.5, 'b': -0.5}
}

# Update all parameters simultaneously without custom loops
# e.g., applying a learning rate update
updated_params = tree_map(lambda x: x - 0.01, params)

print(updated_params)
# {'layer1': [0.99, 1.99, 2.99], 'layer2': {'w': 0.49, 'b': -0.51}}
    \end{lstlisting}
\end{frame}

% ================= SECTION 6: Conclusion ================= %
\section{Conclusion}

\begin{frame}{Summary: The Power of Composable Transformations}
    JAX transforms the way we write mathematical code by allowing arbitrary composition of its core functions:

    \vspace{0.3cm}
    \begin{block}{The Ultimate JAX Pattern}
        \centering
        \texttt{fast\_batched\_gradient\_descent = jit(vmap(grad(loss\_fn)))}
    \end{block}

    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{\texttt{grad}:} Computes the derivative for a single example.
        \item \textbf{\texttt{vmap}:} Vectorizes the gradient computation to process an entire batch.
        \item \textbf{\texttt{jit}:} Compiles the entire batched gradient update into a single, highly optimized GPU/TPU kernel.
    \end{itemize}
\end{frame}
